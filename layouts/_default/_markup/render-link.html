{{- $destination := .Destination -}} {{- $siteRoot := strings.TrimSuffix "/" site.BaseURL -}} {{- $sameSiteHTTPS := or (eq $destination $siteRoot) (strings.HasPrefix $destination (printf "%s/" $siteRoot)) (strings.HasPrefix $destination (printf "%s?" $siteRoot)) (strings.HasPrefix $destination (printf "%s#" $siteRoot)) -}} {{- $siteRootHTTP := replace $siteRoot "https://" "http://" -}} {{- $sameSiteHTTP := or (eq $destination $siteRootHTTP) (strings.HasPrefix $destination (printf "%s/" $siteRootHTTP)) (strings.HasPrefix $destination (printf "%s?" $siteRootHTTP)) (strings.HasPrefix $destination (printf "%s#" $siteRootHTTP)) -}} {{- $isWebURL := or (strings.HasPrefix $destination "http://") (strings.HasPrefix $destination "https://") (strings.HasPrefix $destination "//") -}} {{- $isSameSite := or $sameSiteHTTPS $sameSiteHTTP -}} {{- if and $isWebURL (not $isSameSite) -}} {{- .Text | safeHTML -}} {{- else -}} {{ .Text | safeHTML }} {{- end -}}
